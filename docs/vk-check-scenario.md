### 0.2. Скрипт

Сохрани как `vk_check.py` в пустую рабочую папку (например `~/vk-check/`) и запускай оттуда. Сырые ответы VK сложатся в подпапку `vk-check-out/`. Токены скрипт нигде не печатает.

```python
#!/usr/bin/env python3
"""Ручная проверка VK API для пункта 14.10. Только стандартная библиотека.

    python3 vk_check.py get [--key service|community|bad] [--count N] [--offset N] [--filter F] [--label L]
    python3 vk_check.py byid ID [ID ...] [--key ...] [--label L]
    python3 vk_check.py byid-range N [--key ...]     # N несуществующих/существующих id подряд: проверка лимита списка
    python3 vk_check.py raw METHOD key=value ... [--key ...] [--label L]

Токены берутся из $DOBROMIR_DATA/.env (по умолчанию ~/dobromir-data/.env) и нигде не печатаются.
Сырые ответы пишутся в ./vk-check-out/<метка>.json.
"""
import argparse, hashlib, json, os, sys, time, urllib.parse, urllib.request
from pathlib import Path

API = "https://api.vk.ru/method/"
V = "5.199"
OWNER = -242263522
OUT = Path("vk-check-out")


def load_env():
    p = Path(os.environ.get("DOBROMIR_DATA", Path.home() / "dobromir-data")) / ".env"
    for line in p.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if line and not line.startswith("#") and "=" in line:
            k, v = line.removeprefix("export ").split("=", 1)
            os.environ.setdefault(k.strip(), v.strip().strip("'\""))


def token(key):
    if key == "bad":
        return "invalid_token_for_test"
    return os.environ["VK_SERVICE_TOKEN" if key == "service" else "VK_COMMUNITY_TOKEN"]


def call(method, key, **params):
    data = urllib.parse.urlencode({**params, "access_token": token(key), "v": V}).encode()
    try:
        with urllib.request.urlopen(urllib.request.Request(API + method, data=data), timeout=30) as r:
            return json.load(r)
    except Exception as e:  # сеть/HTTP: это НЕ ответ VK
        return {"transport_error": f"{type(e).__name__}: {e}"}


def save(label, body):
    OUT.mkdir(exist_ok=True)
    (OUT / f"{label}.json").write_text(json.dumps(body, ensure_ascii=False, indent=2), encoding="utf-8")


def short(s):
    return hashlib.sha1(s.encode()).hexdigest()[:8]


def att_ids(p):
    """Только идентичность вложений, без ссылок (ссылки на фото могут содержать подпись, которая меняется)."""
    out = []
    for a in p.get("attachments", []):
        o = a.get(a["type"], {})
        out.append(f'{a["type"]}:{o.get("owner_id")}_{o.get("id")}')
    return out


def hashes(p):
    content = json.dumps([p.get("text"), p.get("edited"), att_ids(p)], ensure_ascii=False)
    full = json.dumps({k: p.get(k) for k in ("text", "edited", "attachments", "copy_history")}, sort_keys=True, ensure_ascii=False)
    everything = json.dumps(p, sort_keys=True, ensure_ascii=False)
    return short(content), short(full), short(everything)


def show_posts(items):
    print("id | дата | pin | edited | тип | текст(симв.) | вложения | likes/views/comments/reposts | h_содержимое h_вложения_полные h_всё")
    for p in items:
        c = lambda k: (p.get(k) or {}).get("count", "-")
        print(" | ".join(map(str, [
            p.get("id"), time.strftime("%d.%m %H:%M", time.localtime(p.get("date", 0))),
            p.get("is_pinned", "-"), p.get("edited", "-"), p.get("post_type"), len(p.get("text", "")),
            (",".join(a["type"] for a in p.get("attachments", [])) or "-")
            + (" +repost" if p.get("copy_history") else ""),
            f'{c("likes")}/{(p.get("views") or {}).get("count", "-")}/{c("comments")}/{c("reposts")}',
            " ".join(hashes(p)),
        ])))


def report(label, body):
    save(label, body)
    if "transport_error" in body:
        print("СЕТЕВАЯ ОШИБКА (не ответ VK):", body["transport_error"]); return
    if "error" in body:
        e = body["error"]
        print(f'ОШИБКА VK: код {e.get("error_code")}: {e.get("error_msg")}'); return
    r = body["response"]
    items = r.get("items") if isinstance(r, dict) else r
    shape = "объект" + str(sorted(r)) if isinstance(r, dict) else "список"
    print(f"ответ: {shape}; постов в ответе: {len(items) if items is not None else '?'}"
          + (f"; count в ответе: {r['count']}" if isinstance(r, dict) and "count" in r else ""))
    if items and isinstance(items[0], dict) and "owner_id" in items[0]:
        show_posts(items)
    print(f"сырой ответ: {OUT / (label + '.json')}")


def main():
    load_env()
    ap = argparse.ArgumentParser()
    ap.add_argument("cmd", choices=["get", "byid", "byid-range", "raw"])
    ap.add_argument("args", nargs="*")
    ap.add_argument("--key", default="service", choices=["service", "community", "bad"])
    ap.add_argument("--count", default="20")
    ap.add_argument("--offset", default="0")
    ap.add_argument("--filter", default="all")
    ap.add_argument("--label")
    a = ap.parse_args()
    label = a.label or f"{a.cmd}-{a.key}-{int(time.time())}"
    if a.cmd == "get":
        body = call("wall.get", a.key, owner_id=OWNER, count=a.count, offset=a.offset, filter=a.filter)
    elif a.cmd == "byid":
        body = call("wall.getById", a.key, posts=",".join(f"{OWNER}_{i}" for i in a.args))
    elif a.cmd == "byid-range":
        n = int(a.args[0])
        body = call("wall.getById", a.key, posts=",".join(f"{OWNER}_{i}" for i in range(1, n + 1)))
    else:
        body = call(a.args[0], a.key, **dict(x.split("=", 1) for x in a.args[1:]))
    report(label, body)


if __name__ == "__main__":
    sys.exit(main())
```

Что умеет:

- `get` - `wall.get` и таблица постов. В колонках: id, дата, закреп, `edited`, тип, длина текста, типы вложений, счетчики (лайки/просмотры/комментарии/репосты) и три коротких хэша. Хэш - это просто «отпечаток» содержимого: если содержимое то же, отпечаток тот же.
  - `h_содержимое` - текст, `edited` и **идентификаторы** вложений (то, что мы хотим хэшировать в воркере);
  - `h_вложения_полные` - то же, но вложения целиком, со всеми ссылками на картинки;
  - `h_всё` - весь объект поста вместе со счетчиками.
- `byid` - `wall.getById` по id постов.
- `byid-range N` - `wall.getById` сразу по id от 1 до N (для проверки лимита списка; несуществующие id VK должен просто пропустить, это тоже полезно записать).
- `raw` - любой другой метод.
- `--key service|community|bad`: какой ключ использовать (`bad` - заведомо неверный, нужен для сравнения «ошибка» и «пусто»).

### 0.4. Подготовь тестовые посты

Нужна стена, где есть минимум 5 обычных постов. Старые тестовые посты (текст, фото, ссылка, документ) на стене уже есть. Добавь руками:

- пост **P1**: текст + 1 картинка (для разделов 1 и 5);
- по ходу работы создашь еще несколько временных постов (D1, D2, D3) для раздела 4.

Записывай id постов (число в адресе поста `vk.ru/wall-242263522_<ID>`).

## 1. Поле `edited` (правка текста и замена картинки)

**Зачем.** Воркер должен заметить, что пост правили. Если `edited` есть и меняется, это самый дешевый признак; хэш содержимого остается страховкой.

Шаги (пост P1, его id = `ID1`):

| Шаг | Что сделать в группе | Запрос | Что записать |
|---|---|---|---|
| 1.1 | Ничего, исходное состояние | `python3 vk_check.py get --count 20 --label e1-start` | Есть ли у P1 поле `edited` (колонка `edited`: `-` значит нет). Три хэша |
| 1.2 | Подожди 1-2 минуты. Измени в тексте одну букву, сохрани | `python3 vk_check.py get --count 20 --label e2-text` | Появилось/изменилось ли `edited` (это Unix-время, сравни с временем правки), изменился ли `h_содержимое` |
| 1.3 | Открой редактирование и сохрани **без изменений** | `python3 vk_check.py get --count 20 --label e3-noop` | Меняется ли `edited` от пустого сохранения. Меняется ли `h_содержимое` (не должен) |
| 1.4 | Замени картинку на другую (удали старую, добавь новую) | `python3 vk_check.py get --count 20 --label e4-photo` | Меняется ли `edited`; меняется ли `h_содержимое` (должен, потому что поменялся id фото) |
| 1.5 | Добавь вторую картинку, потом убери ее | `get` с метками `e5-add`, `e6-remove` | То же. Нужно понять, ловит ли хэш по идентификаторам вложений любое изменение набора картинок |
| 1.6 | Только ради контроля: запроси тот же пост через `getById` | `python3 vk_check.py byid ID1 --label e7-byid` | Совпадают ли `edited` и хэши с тем, что дал `wall.get` (важно: воркер сверяет через `getById`, а ищет изменения через `wall.get`) |

| Что | Ожидалось | Получено |
|---|---|---|
| Поле `edited` есть у неправленного поста | нет [предположение] | |
| `edited` после правки текста | появилось, равно времени правки [предположение] | |
| `edited` после «сохранить без изменений» | не меняется или меняется (оба варианта возможны) | |
| `edited` после замены картинки | меняется [предположение] | |
| `h_содержимое` после замены картинки | меняется (меняется id фото) | |
| `edited` и хэши в `wall.get` и `getById` | совпадают | |

**Как влияет на воркер.** Хэш считаем по тексту, `edited` и идентификаторам вложений. Если `edited` при замене картинки не меняется, хэш все равно поймает правку по идентификаторам вложений. Если `edited` меняется от «сохранить без изменений» (шаг 1.3), сайт будет лишний раз пересобираться и писать «Обновлено...»; тогда `edited` из хэша убираем и опираемся только на текст и вложения, а сам `edited` используем лишь для подписи «Обновлено» при реальной правке.

## 2. Максимальный `count` и размер списка в `getById`

**Зачем.** Нужно знать, сколько постов за запрос можно запрашивать при сверке всей базы и какое окно безопасно.

**Ограничение теста.** Постов в группе мало (17-30), поэтому «отдал сколько есть» нельзя отличить от «тихо обрезал до 100». Уверенно проверяется только поведение, когда лимит превышен: ошибка или нет. Максимум по справочнику - 100 (`wall.get`), `getById` - до 100 id [справочник, но не проверено на живом API].

```sh
python3 vk_check.py get --count 100 --label c-100
python3 vk_check.py get --count 101 --label c-101
python3 vk_check.py get --count 1000 --label c-1000
python3 vk_check.py get --count 0 --label c-0
python3 vk_check.py get --count 100 --key community --label c-100-community
python3 vk_check.py byid-range 100 --label g-100
python3 vk_check.py byid-range 101 --label g-101
python3 vk_check.py byid-range 200 --label g-200
```

`byid-range N` запрашивает id от 1 до N: существующие вернутся, несуществующие VK должен пропустить (это заодно проверка шага 4.4). Если у твоих постов id больше 100, команда вернет пусто, это нормально: подставь для проверки лимита подходящий диапазон (поправь `range(1, n + 1)` в скрипте на `range(START, START + n)`).

| Что | Ожидалось | Получено |
|---|---|---|
| `wall.get` `count=100` (сервисный ключ) | ок | |
| `count=101` | ошибка 100 или тихое обрезание до 100 [предположение] | |
| `count=1000` | то же | |
| `count=0` | пусто или ошибка [предположение] | |
| В ответе `wall.get` есть общий `count` (сколько всего постов) | да | |
| `wall.get` ключом сообщества | ошибка метода/доступа (см. 0.3) | |
| `getById` на 100 id | ок | |
| `getById` на 101 id | ошибка [предположение] | |
| `getById` на 200 id | ошибка [предположение] | |
| Форма ответа `getById` (список или объект с `items`) | в схеме 5.199 может быть и так, и так [предположение] | |

**Как влияет на воркер.** Окно `VK_WINDOW=50` укладывается в 100 с запасом, менять его не надо, если лимит подтвердился. Сверку всей базы через `getById` режем на пачки по подтвержденному максимуму (если 101 дало ошибку, пачка 100 безопасна; если нет, берем пачку 50 «с запасом», пока не проверим на стене больше 100 постов). Форму ответа `getById` парсер обязан понимать в обоих видах (список и `{"items": [...]}`), либо по результату этой проверки жестко один.

## 4. Удаленный, отложенный, скрытый пост и «пусто против ошибки»

## 7. Сколько CP стоит сборка и как часто пересобирать

Эта проверка не про VK, но от нее зависит `MAX_BUILDS_PER_DAY`. Сейчас там стоит 4, а посты могут появляться каждый час, раз в сутки или не появляться вовсе. При опросе раз в час это до 24 поводов собрать сайт в сутки.

**Что происходит при лимите 4 и частых постах (логика, а не измерение).** Изменения не пропадают, если воркер хранит в базе признак «сайт устарел» (флаг или счетчик несобранных изменений), а не сравнивает с «прошлым запуском». Тогда, когда лимит исчерпан, воркер обновляет базу, но сборку откладывает до следующего свободного слота, и одна сборка забирает все накопленное. Цена: сайт отстает от VK на время ожидания (с лимитом 4 в сутки в худшем случае до 6 часов между сборками). Если же воркер хранит только «что изменилось с прошлого запуска», то при пропущенной сборке эти изменения теряются до следующего изменения. В воркере так и сделано: несобранные изменения ждут в базе со статусом `pending`, это покрыто тестами.

**Как измерить стоимость одной сборки.**

1. Выбери тихое время. Запиши в панели Beget значение CP за текущие сутки (раздел статистики нагрузки, график CP) и время.
2. Выполни сборку тем способом, которым ее будет запускать воркер, через `time`:

```sh
cd ~/a978879k.beget.tech   # или где лежит сайт на хостинге
time npm run build
```

   Запиши `real`, `user`, `sys`. Раньше пустой сайт собирался за real 2,4 с, user 5,7 с, sys 3,1 с. Реальный сайт с новостями и картинками будет заметно тяжелее.
3. Повтори 3 раза подряд. Через 10-30 минут посмотри в панели, насколько вырос CP за сутки (панель обновляется не мгновенно, ее задержка неизвестна [предположение]). Раздели прирост на число сборок. Это стоимость одной сборки в CP.
4. Отдельно замерь холостой запуск воркера (без изменений, только `wall.get` и сверка): `time deploy/run-worker.sh` и прирост CP за 5-10 запусков (24 запуска в сутки).

| Что | Получено |
|---|---|
| `time npm run build` (real / user / sys), 3 прогона | |
| Прирост CP на одну сборку | |
| `time` холостого запуска воркера | |
| Прирост CP на холостой запуск | |
| Фоновое потребление сайта за сутки (старый WP давал 2-6 CP) | |

Формула для выбора лимита: `MAX_BUILDS_PER_DAY ≈ (80 − фон − 24 × стоимость холостого запуска) × 0,5 / стоимость сборки`. Множитель 0,5 это запас, чтобы не упереться в блокировку по CP. Например, при сборке в 3 CP, холостом запуске в 0,1 CP и фоне 6 CP получится `(80 − 6 − 2,4) × 0,5 / 3 ≈ 12`, то есть до 12 сборок в сутки. Если сборка стоит 15 CP, получится около 2. Цифры в примере выдуманные, подставь свои.
